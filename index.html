Step-by-Step Google Sheet Setup

  1. Create a new Google Sheet and name Row 1 headers: title, tmdb_id, type, rewatches
  2. Click Extensions → Apps Script and paste the script below:
  3. Click Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone).
function doGet() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const rows = sheet.getDataRange().getValues();
  const headers = rows.shift();
  const data = rows.map(r => ({
    title: r[0],
    tmdb_id: r[1],
    type: r[2] || 'movie',
    rewatches: Number(r[3] || 1)
  }));
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const body = JSON.parse(e.postData.contents);
  sheet.appendRow([body.title, body.tmdb_id, body.type, body.rewatches || 1]);
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
